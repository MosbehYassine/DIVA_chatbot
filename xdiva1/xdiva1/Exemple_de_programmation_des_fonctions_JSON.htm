Exemple de programmation des fonctions JSON

 

procedure TraiterErreurJson(ticket,st)

1 st L

1 merr S

1 mexcept S

1 ticket L

beginp

   switch st

   case 1

      MessageBox("ticket inconnu"," ")

   case 2

      JsonGetError(ticket,merr,mexcept)

      MessageBox(mexcept,merr)

   case 3

      MessageBox("Erreur de syntaxe dans l'expressions","")

   endswitch

endp

 

 

1 ticket L

1 json   S

1 valeur S

1 st     L

1 nbindices L

1 i       L

1 merr    S

1 mexcept S

1 j       L

 

main

json = "{ 'menu': {" & \

                           "'id': 'file',"  & \

                           "'value': 'File',"  & \

                           "'popup': {"  & \

                           "'menuitem': ["  & \

                              "{'value': 'New', 'onclick': 'CreateNewDoc()'},"  & \

                              "{'value': 'Open', 'onclick': 'OpenDoc()'},"  & \

                              "{'value': 'Close', 'onclick': 'CloseDoc()'}"  & \

                           "]"  & \

                           "}"  & \

                           "}}";

 

ticket = JsonOpen()

st = JsonParse(ticket,json)

TraiterErreurJson(ticket,st)

 

st = JsonGetValue(ticket,"menu.id",valeur);                         --> file

TraiterErreurJson(ticket,st)

 

st = JsonGetValue(ticket,"menu.value",valeur);                      -->File

TraiterErreurJson(ticket,st)

 

st = JsonGetValue(ticket,"menu.popup.menuitem[1].onclick",valeur);  -->CreateNewDoc()

TraiterErreurJson(ticket,st)

 

st = JsonGetLength(ticket, "menu.popup.menuitem", nbindices);       --> 3

TraiterErreurJson(ticket,st)

 

; balayage de la liste des menuitem

for i = 1 to nbindices

    st =  JsonGetValue(ticket, "menu.popup.menuitem[" & tostring(i) & "].value", valeur);

    TraiterErreurJson(ticket,st)

    st =  JsonGetValue(ticket, "menu.popup.menuitem[" & tostring(i) & "].onclick",valeur);

    TraiterErreurJson(ticket,st)

next

 

JsonClose(ticket)
