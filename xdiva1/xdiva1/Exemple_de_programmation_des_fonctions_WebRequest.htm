define mp = "monmotdepasse"

 

procedure TraiterErreurJson(ticket,st)

;* affichage d'une erreur sur les fonctions JSON

1 st L

1 merr S

1 mexcept S

1 ticket L

beginp

   switch st

   case 1

      MessageBox("ticket inconnu"," ")

   case 2

      JsonGetError(ticket,merr,mexcept)

      MessageBox(mexcept,merr)

   case 3

      MessageBox("Erreur de syntaxe dans l'expressions","")

   endswitch

endp

 

procedure TraiterErreurRequete(ticket,st)

;* affichage d'une erreur sur les fonctions WebRequest

1 st L

1 merr S

1 mexcept S

1 err L

1 ticket L

beginp

   switch st

   case 1

      MessageBox("ticket inconnu"," ")

   case 2

      WebRequestGetError(ticket,merr,mexcept,err)

      MessageBox(mexcept,merr & " Erreur web=" & tostring(err))

   endswitch

endp

 

main

1 ticketJson L

1 st L

1 ticketRequete L

1 retour S

1 nbelts L

1 valeur S

1 statusCode L

1 i L

 

   ; alloction de la requête

   ticketRequete = WebRequestNew()

 

   ; garnissage des paramètres

   st = WebRequestSetUrl(ticketRequete,"https://jsonplaceholder.typicode.com/posts")

   TraiterErreurRequete(ticketRequete,st)

 

   ; uniquement si on passe par un proxy

   st = WebRequestSetProxy(ticketRequete,"http://10.10.10.123:8080","DOMAINE","UTILISATEUR",mp)

   TraiterErreurRequete(ticketRequete,st)

 

   st = WebRequestSetMethod(ticketRequete,"GET",-1)

   TraiterErreurRequete(ticketRequete,st)

   st = WebRequestSetHeader(ticketRequete,"Content-Type","application/json");

   TraiterErreurRequete(ticketRequete,st)

 

   ; exécution de la requête

   st = WebRequestSend(ticketRequete , "" , retour, statusCode)

   TraiterErreurRequete(ticketRequete,st)

   ; libération de la requête

   st = WebRequestClose(ticketRequete,"close")

   TraiterErreurRequete(ticketRequete,st)

 

   ;* traitement de la réponse

   ; allocation d'un objet json

   ticketJson = JsonOpen(1000)

   ; création de l'objet Json depuis la chaîne reçue

   st = JsonParse(ticketJson,retour);

   TraiterErreurJson(ticketJson,st)

 

;[

;  },

;]

   ; recupération du nombre d'éléments du tableau

   st= JsonGetLength(ticketJson,"",nbelts)   ; le json commence directement par [

   TraiterErreurJson(ticketJson,st)

   ; recupération du titre du premier élément

   st= JsonGetValue(ticketJson,"[1].title",valeur)

   TraiterErreurJson(ticketJson,st)

   ; balayage de tous les éléments

   for i = 1 to nbelts

      st= JsonGetValue(ticketJson,"[" & tostring(i)  & "].title",valeur)

      TraiterErreurJson(ticketJson,st)

      st= JsonGetValue(ticketJson,"[" & tostring(i)  & "].body",valeur)

      TraiterErreurJson(ticketJson,st)

   next

   ; libération de l'objet JSON

   JsonClose(ticketJson)

   ProgramExit
