Exemple de programmation d'un DVBuffer

 

1 emission L

1 reception L

1 numeroAction L = 1234

1 numeroActionRecue L

1 parametreRecu S

1 tout S

1 prop X

main

const propriete_debut     = 0x01

const propriete_fin      = 0x02

const propriete_action    = 0x03

const propriete_parametre = 0x04

emission = DVBufferGetId() ; création du DVBuffer

DVBufferPut(emission,propriete_debut)     ; propriété sans valeur

DVBufferPut(emission,propriete_action)    ; propriété avec valeur L

DVBufferPut(emission,numeroAction)

DVBufferPut(emission,propriete_parametre) ; propriété avec valeur alpha

DVBufferPut(emission,"parametres")

DVBufferPut(emission,propriete_fin)

DVBufferToString(emission,tout)           ; récupération du buffer et transmission

DVBufferDestroy(emission)

 

; ...... reception du DVBuffer

 

reception = DVBufferGetId(chaine = tout)   ; création d'un DVBuffer en lui passant la chaine

DVBufferGet(reception,prop)

loop prop <> propriete_fin                 ; récupération des valeurs

switch prop

case propriete_debut

case propriete_action

DVBufferGet(reception,numeroActionRecue)

case propriete_parametre

DVBufferGet(reception,parametreRecu)

default

MessageBox("Propriété inconnue"," ")

endswitch

DVBufferGet(reception,prop)

endloop

DVBufferDestroy(reception)

ProgramExit
