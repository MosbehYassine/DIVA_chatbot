FtpRequestExecRequest Famille Thème

 

Exécution d'une commande.

 

int FtpRequestExecRequest  ( Ticket , &Retour, &Description)

 

Paramètres

Ticket

Identifiant de la requête obtenu par FtpRequestNew

 

Retour

Données renvoyées par le serveur. Par exemple la taille d'un fichier ou la liste d'un répertoire.

 

Description

Chaîne de caractères indiquant le stauts renvoyé par le serveur.

 

Retour

0

Pas d'erreur

1

Ticket incorrect

2

Exception. Voir FtpRequestGetError

 

Exemple

A titre d'exemple, le code source de la fonction FtpDownloadFile.

 

;*

Public Function String_FtpResult FtpDownloadFile(server, login, password, localFile, remoteFile, OpenHarmony, proxyServer, ProxyMode, proxyLogin, proxyPassword)

; Téléchargement d'un fichier depuis un serveur FTP

1 server        S ; Adresse du serveur : ftp://ip[:port]  ou  sftp://ip[:port]

1 login         S ; Identifiant sur le serveur. 'anonymous' si vide

1 password      S ; Mot de passe sur le serveur

1 localFile     S ; Chemin d'accès au fichier sur l'ordinateur local

1 remoteFile    S ; Chemin d'accès au fichier sur le serveur distant

1 OpenHarmony   1,0 = FALSE ; Ouverture du fichier en local en mode Harmony

1 proxyServer   S = ' ' ; Adresse du proxy

1 ProxyMode     1,0 = 1 ; Mode d'identification sur le proxy. 1 : Utiliser le proxy du système ; 2 : SSPI ; 3 : configuration manuelle

1 proxyLogin    S = ' ' ; Identifiant sur le serveur proxy

1 proxyPassword S = ' ' ; Mot de passe sur le serveur proxy

1 id L

1 st L

1 retour S

1 desc S

1 mess S

1 ex S

RecordDiva FtpResult

 

BeginF

   FtpResult = ""

   if server = '' or localFile = '' or remoteFile = ''

      FtpResult.Status = 3

      FReturn FtpResult

   endif

   id = FtpRequestNew(condition(left(server, 1) = 's', true, false))

   st = FtpRequestSetUrl(id, server, remoteFile)

   if st <> 0

      FtpRequestGetError(id, mess, ex)

      FtpResult.Status = st

      FtpResult.Retour = mess

      FtpResult.Desc = ex

      FtpRequestClose(id)

      freturn FtpResult

    endif

    st = FtpRequestSetLogin(id, login, password)

    if st <> 0

       FtpRequestGetError(id, mess, ex)

       FtpResult.Status = st

       FtpResult.Retour = mess

       FtpResult.Desc = ex

       FtpRequestClose(id)

       freturn FtpResult

    endif

    if proxyServer <> ' '

       st = FtpRequestSetProxy(id, proxyServer, ProxyMode, proxyLogin, ProxyPassword)

       if st <> 0

           FtpRequestGetError(id, mess, ex)

           FtpResult.Status = st

           FtpResult.Retour = mess

           FtpResult.Desc = ex

           FtpRequestClose(id)

           freturn FtpResult

       endif

    endif

    st = FtpRequestSetMethod(id, "RETR", localFile, OpenHarmony)

    if st <> 0

        FtpRequestGetError(id, mess, ex)

        FtpResult.Status = st

        FtpResult.Retour = mess

        FtpResult.Desc = ex

        FtpRequestClose(id)

        freturn FtpResult  

    endif

    st = FtpRequestExecRequest(id, retour, desc)

    FtpResult.Status = st

    FtpResult.Retour = retour

    FtpResult.Desc = desc

    if st <> 0

        FtpRequestGetError(id, mess, ex)

        FtpResult.Status = st

        FtpResult.Retour = mess

        FtpResult.Desc = ex

        FtpRequestClose(id)

        freturn FtpResult

    endif

    FtpRequestClose(id)

    freturn FtpResult

EndF
