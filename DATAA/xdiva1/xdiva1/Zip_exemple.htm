Fonctions Zip Exemple

 

module "dhzip.dhop"

1 file1 260

1 file2 260

1 file3 260

1 err L

1 fd L

1 lg L

1 texte S = "ceci est un petit exemple en zip"

1 lgdest L

1 lgdest2 L

1 ss S

1 ss2 S

1 expanse S = "1234567890"

1 comprime S

1 verif S

 

 

file1 = "C:\divalto\Test\test.dhsp"

file2 = "C:\divalto\Test\testCompresse.zip"

 

err = ZipOpen

if err

   display "erreur " & tostring(err)

   programexit

endif

 

; compression de fichier

err = ZipComprFiles(left(file1),left(file2),1)

if err

   display "erreur " & tostring(err)

endif

 

; décompression de fichier

; il faut que le fichier existe !

file1 = "C:\divalto\Test\testDecompresse.dhsp"

err = ZipUncomprFiles(left(file2),left(file1),1)

if err

   display "erreur " & tostring(err)

endif

 

; compression en mémoire

comprime = ZipComprMemory(expanse,size(expanse),lgdest)

comprime = left(comprime,lgdest)

 

; décompression en mémoire

verif = ZipUnComprMemory(comprime,size(comprime),lgdest)

verif = left(verif,lgdest)

 

ZipClose
