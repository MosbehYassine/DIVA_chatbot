Cas d'utilisation

 

  • Déclaration commune aux cas d'utilisation

; Déclaration des chemins vers les clés

1 pathFilenamePrivateKey S

1 pathFilenamePublicKey S

; Déclaration des clés

1 privateKey L

1 publicKey L

; Déclaration d'un message à signer, de la signature et un chemin vers un fichier

1 msg S

1 signature S

1 bigFile S

; Valeur de retour et code d'erreur

1 retVal B

1 errorCode L

 

pathFilenamePrivateKey = "C:/tmp/privateKey.txt"

pathFilenamePublicKey = "C:/tmp/publicKey.txt"

bigFile = "C:/tmp/unGrosFichier.dat"

  • Création initiale des clés (dans des fichiers) avec traitement des erreurs

; *** Clé privé ***

; Création d'une clé privée dans un fichier

retVal = PrivateKeyCreateToFile(pathFilenamePrivateKey)

if retVal = 0

errorCode = CryptographyGetLastErrorCode()

display CryptographyGetErrorMessage(errorCode)

ProgramExit()

endif

 

; Création de la clé privée à partir de ce même fichier

privateKey = PrivateKeyCreateFromFile(pathFilenamePrivateKey)

if privateKey = 0

errorCode = CryptographyGetLastErrorCode()

display CryptographyGetErrorMessage(errorCode)

ProgramExit()

endif

 

; *** Clé publique ***

; Création d'une clé publique dans un fichier

retVal = PublicKeyCreateToFile(privateKey, pathFilenamePublicKey)

if retVal = 0

; todo: traiter l'erreur

endif

; Création de la clé privée à partir de ce même fichier

publicKey = PublicKeyCreateFromFile(pathFilenamePublicKey)

if publicKey = 0

; todo: traiter l'erreur

endif

 

; Destruction des clés

retVal = PrivateKeyDestroy( privateKey )

retVal = PublicKeyDestroy( publicKey )

  • Signature d'un message

; Par souci de clarté, le traitement des erreurs n'est pas écrit (voir le premier cas d'utilisation).

 

; Chargement des clés

privateKey = PrivateKeyCreateFromFile(pathFilenamePrivateKey)

publicKey = PublicKeyCreateFromFile(pathFilenamePublicKey)

 

; Signature d'un message

msg = "mon message à signer"

signature = PrivateKeySignMessage( privateKey, msg )

if Length(signature) = 0

; todo: traiter l'erreur

endif

 

; Vérification de la signature d'un message

retVal = PublicKeyVerifyMessageSignature( publicKey, msg, signature )

if retVal = 0

display "Echec de la vérification de la signature"

; todo: traiter l'erreur

endif

 

; Destruction des clés

retVal = PrivateKeyDestroy( privateKey )

retVal = PublicKeyDestroy( publicKey )

  • Signature d'un fichier

Similaire à l'exemple précédent en remplaçant PrivateKeySignMessage par PrivateKeySignFile (respectivement PublicKeyVerifyMessageSignature par PublicKeyVerifyFileSignature).
