;Schéma de programmation d'onglets en saisie de masque

...

 

fini = 3

NumPage = 1                                  ;page initiale

 

LOOP fini = 3

   fini = 0                                  ;saisie de la page NumPage 

   XmeInput ("Masque",NumPage,0,XME_MODE_GOTO) 

   DO

      SWITCH Harmony.Arret 

         CASE Harmony.Key = K_ESCAPE 

            fini = 1                         ;abandon 

 

         CASE 200                            ;arrêt onglets : 

            ...                              ;traitements suivant la

            ...                              ;validation de la page 

            NumPage = Harmony.SourisPosX     ;changement de page 

            fini = 3 

 

         ...                                 ;autres cas 

 

                                             ;après les points d'arrêt 

         CASE Harmony.Key in (K_RETURN, K_TAB, K_DOWN)

            ...                              ;traitements suivant la

            ...                              ;validation de la page 

            fini = 2

      ENDSWITCH 

   WHILE fini = 0 

      XMENEXT (Harmony.Retour, Harmony.CplRetour)

   WEND 

ENDLOOP
