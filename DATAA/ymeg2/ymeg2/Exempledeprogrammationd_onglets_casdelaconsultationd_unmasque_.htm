Exemple de programmation d'onglets (cas de la consultation d'un masque)

;Schéma de programmation d'onglets en consultation

;(par XmeConsult ou XmeListConsult)

...

 

fini = 0

NumPage = 1                           ;page initiale

 

LOOP fini = 0

   XMECONSULT ("Masque", NumPage, XME_MODE_GOTO)

   IF Harmony.DataArret = 200         ;point d'arrêt onglets :

      NumPage = Harmony.SourisPosX    ;changement de page 

   ELSE                               ;frappe d'une touche : 

      fini = 1                        ;sortie 

   ENDIF 

ENDLOOP
