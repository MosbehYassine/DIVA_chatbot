<RecordSqlAttributeName> ::=

   { DataBase | comment | ZoomOptimize }

   

<SelectAttributeName> ::=

   { Top=<n> | Distinct= { yes | no } }

   

<ColumnList> ::=

   { { { table | alias }.columnName [ as uniqueName ] [ Name=name ] [ NoActivate ] [ GroupBy=yes|no ] } &|

     { Expression as uniqueName [ Name=name ] [ NoActivate ] [ GroupBy=yes|no ] } &|

     { { table | alias }.* }

   }[ ,...n ]

   

<JoinExpression> ::=

   { Join | LeftJoin | RightJoin | FullJoin | CrossJoin | Join01 }

   

<Condition> ::=

   { = | <> | <= | >= | in | not in | is null | is not null | like | not like }

   

<ConditionText> ::=

   {

      { { table | alias }.columnName | Expression } <Condition> {  { { table | alias }.columnName } | value | Expression }

   }[ ,...n ]

   

   

<ParamType> ::=

   { char | int | date }

<OrderByText> ::=

   {

      { table | alias }.columnName { ASC | DESC }

   }[ ,...n ]

 

<GroupByAttributeName> ::=

   { Dynamic= { yes | no } }
